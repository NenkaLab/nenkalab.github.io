// SHA256 해싱
async function sha256(message) {
    const msgBuffer = new TextEncoder().encode(message);
    const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray;
}

// XOR 복호화
function xorDecrypt(base64String, passwordHash) {
    const data = atob(base64String);
    const dataBytes = Array.from(data).map(c => c.charCodeAt(0));
    
    // 비밀번호 해시를 반복해서 데이터 길이만큼 확장
    const keyBytes = [];
    for (let i = 0; i < dataBytes.length; i++) {
        keyBytes.push(passwordHash[i % passwordHash.length]);
    }
    
    // XOR 연산 (암호화와 동일)
    const decrypted = dataBytes.map((byte, i) => byte ^ keyBytes[i]);
    
    // 다시 base64로 인코딩
    const decryptedString = String.fromCharCode(...decrypted);
    return btoa(decryptedString);
}

// Base64 디코딩
async function decodeBase64() {
    let input = inputText.value.trim();
    
    if (!input) {
        outputText.value = '';
        return;
    }
    
    try {
        // 비밀번호 복호화 적용
        if (passwordEnabled && passwordInput.value) {
            const passwordHash = await sha256(passwordInput.value);
            input = xorDecrypt(input, passwordHash);
        }
        
        // 인코딩에 따라 처리
        let decoded;
        if (selectedEncoding === 'utf-8') {
            decoded = decodeURIComponent(escape(atob(input)));
        } else if (selectedEncoding === 'utf-16') {
            const bytes = atob(input).split('').map(c => c.charCodeAt(0));
            const utf16Array = [];
            for (let i = 0; i < bytes.length; i += 2) {
                const code = bytes[i] | (bytes[i + 1] << 8);
                utf16Array.push(String.fromCharCode(code));
            }
            decoded = utf16Array.join('');
        } else if (selectedEncoding === 'iso-8859-1' || selectedEncoding === 'ascii') {
            decoded = atob(input);
        }
        
        outputText.value = decoded;
        
        // 자동 디코딩이 켜져있으면 5초 후 기록 저장
        if (autoDecodeEnabled) {
            clearTimeout(autoDecodeTimeout);
            autoDecodeTimeout = setTimeout(() => {
                saveToHistory(inputText.value, decoded);
            }, 5000);
        }
    } catch (error) {
        alert('디코딩 중 오류가 발생했습니다.\n올바른 Base64 문자열인지, 비밀번호가 맞는지 확인하세요.');
        outputText.value = '';
    }
}

    }, 1000);
}

}

// 기록 저장
function saveToHistory(input, output) {
    const history = JSON.parse(localStorage.getItem('decodeHistory') || '[]');
    
    const item = {
        id: Date.now(),
        input: input.substring(0, 50) + (input.length > 50 ? '...' : ''),
        output: output.substring(0, 50) + (output.length > 50 ? '...' : ''),
        fullInput: input,
        fullOutput: output,
        timestamp: new Date().toLocaleString('ko-KR'),
        encoding: selectedEncoding,
        hasPassword: passwordEnabled
    };
    
    history.unshift(item);
    
    // 최대 20개까지만 저장
    if (history.length > 20) {
        history.pop();
    }
    
    localStorage.setItem('decodeHistory', JSON.stringify(history));
    loadHistory();
}

// 기록 불러오기
function loadHistory() {
    const history = JSON.parse(localStorage.getItem('decodeHistory') || '[]');
    
    if (history.length === 0) {
        historyList.innerHTML = '<p class="text-sm text-zinc-500 dark:text-zinc-400 text-center py-4">변환 기록이 없습니다</p>';
        return;
    }
    
    historyList.innerHTML = history.map(item => `
        <div class="history-item p-3 bg-zinc-50 dark:bg-zinc-900 rounded-lg border border-zinc-200 dark:border-zinc-700 cursor-pointer"
             onclick="loadFromHistory(${item.id})">
            <div class="flex items-start justify-between mb-2">
                <span class="text-xs text-zinc-500 dark:text-zinc-400">${item.timestamp}</span>
                <div class="flex gap-2">
                    ${item.hasPassword ? '<span class="text-xs bg-yellow-100 dark:bg-yellow-900 text-yellow-800 dark:text-yellow-200 px-2 py-0.5 rounded">🔒</span>' : ''}
                    <span class="text-xs bg-green-100 dark:bg-green-900 text-green-800 dark:text-green-200 px-2 py-0.5 rounded">${item.encoding}</span>
                </div>
            </div>
            <p class="text-sm text-zinc-700 dark:text-zinc-300 font-mono truncate mb-1">입력: ${item.input}</p>
            <p class="text-sm text-zinc-500 dark:text-zinc-400 font-mono truncate">출력: ${item.output}</p>
        </div>
    `).join('');
}

// 기록에서 불러오기
function loadFromHistory(id) {
    const history = JSON.parse(localStorage.getItem('decodeHistory') || '[]');
    const item = history.find(h => h.id === id);
    
    if (item) {
        inputText.value = item.fullInput;
        outputText.value = item.fullOutput;
        encodingSelect.value = item.encoding;
        selectedEncoding = item.encoding;
    }
}
